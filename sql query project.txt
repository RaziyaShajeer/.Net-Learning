CREATE TABLE MyUser (
	UserId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
	FirstName VARCHAR(50) NOT NULL,
	Lastname VARCHAR(50) NULL,
	Email VARCHAR(50) NOT NULL,
	Phone VARCHAR(20) NOT NULL,
	Address VARCHAR(100) NOT NULL,
	State VARCHAR(10) NOT NULL,
	Password VARCHAR(10) NOT NULL,
	Role INT NOT NULL,
	LocationId UNIQUEIDENTIFIER NOT NULL,
	CreatedAt DATETIME NOT NULL,
	CONSTRAINT FK_MyUser_Location FOREIGN KEY (LocationId) REFERENCES Location(LocationId)
	);


CREATE TABLE Location (
    LocationId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    LocationName VARCHAR(10) NOT NULL
	);


CREATE TABLE RestaurantProfile (
    RestaurantId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    RestaurantName VARCHAR(50) NOT NULL,
    RestauratType INT NOT NULL,
    Phone VARCHAR(10) NOT NULL,
    LocationId UNIQUEIDENTIFIER NOT NULL,
    RestaurantImage VARBINARY(MAX) NULL,
    Status INT NOT NULL,
    CreatedAt DATETIME NOT NULL,
    CONSTRAINT FK_RestaurantProfile_Location FOREIGN KEY (LocationId) REFERENCES Location(LocationId)
);

CREATE TABLE RestaurantAdmin (
    RestaurantAdminId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    RestaurantId UNIQUEIDENTIFIER NOT NULL,
    CONSTRAINT FK_RestaurantAdmin_RestaurantProfile FOREIGN KEY (RestaurantId) REFERENCES RestaurantProfile(RestaurantId)
);


CREATE TABLE OrderItem (
    OrderItemId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    orderId UNIQUEIDENTIFIER NOT NULL,
    TotalAmount DECIMAL(18, 0) NOT NULL,
    CreatedAt DATETIME NULL,
    CONSTRAINT FK_OrderItem_Order FOREIGN KEY (OrderItemId) REFERENCES MyOrder(OrderId)
);


CREATE TABLE MyOrder (
    OrderId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    UserId UNIQUEIDENTIFIER NOT NULL,
    CreatedAt DATETIME NOT NULL,
    CONSTRAINT FK_Order_User FOREIGN KEY (UserId) REFERENCES MyUser(UserId)
);


CREATE TABLE Dish (
    DishId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    DishName VARCHAR(50) NOT NULL,
    Description VARCHAR(MAX) NOT NULL,
    DishImage VARBINARY(MAX) NULL,
    Category INT NOT NULL,
    Availablity INT NOT NULL,
    Createdat DATETIME NOT NULL,
    RestaurantId UNIQUEIDENTIFIER NOT NULL,
    Price DECIMAL(18, 0) NULL,
    CONSTRAINT FK_Dish_RestaurantProfile FOREIGN KEY (RestaurantId) REFERENCES RestaurantProfile(RestaurantId)
);


CREATE TABLE Cartitem (
    CartItemId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    Cartid UNIQUEIDENTIFIER NOT NULL,
    DishId UNIQUEIDENTIFIER NOT NULL,
    Quanity INT NOT NULL,
    Price DECIMAL(18, 0) NOT NULL,
    Total DECIMAL(18, 0) NOT NULL,
    CONSTRAINT FK_Cartitem_Dish FOREIGN KEY (DishId) REFERENCES Dish(DishId)
);


CREATE TABLE Cart (
    CartId UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    UserId UNIQUEIDENTIFIER NOT NULL,
    CreatedAt DATETIME NOT NULL,
    UpdatedAt DATETIME NOT NULL,
    TotalAmount DECIMAL(18, 0) NOT NULL,
    CONSTRAINT FK_Cart_User FOREIGN KEY (UserId) REFERENCES MyUser(UserId)
);
